  // ── Recurrentes ──────────────────────────────────────────
  async function loadRecurrentes() {
    const ahora = new Date();
    const mes   = ahora.getMonth() + 1;
    const anio  = ahora.getFullYear();
    try {
      const res = await fetch(`${WORKER_URL}/api/recurrentes?mes=${mes}&anio=${anio}`);
      if (!res.ok) return;
      const data = await res.json();
      if (!data.ok) return;

      const section = document.getElementById('recurrentes-section');
      const container = document.getElementById('recurrentes-warn');
      section.style.display = 'block';

      let html = '';

      // Mostrar faltantes primero (advertencias)
      for (const f of data.faltantes) {
        html += `<div class="rec-warn" title="Sin registro en ${mesNombre(mes)} ${anio}">
          <div class="rec-dot"></div> ${f.nombre}
        </div>`;
      }

      // Mostrar confirmados
      for (const o of data.ok) {
        const valor = o.registro ? formatCOP(o.registro.total) : '';
        html += `<div class="rec-ok" title="${valor}">
          <div class="rec-dot"></div> ${o.nombre}
        </div>`;
      }

      // Resumen
      if (data.faltantes.length > 0) {
        html += `<div class="rec-summary">${data.faltantes.length} de ${data.faltantes.length + data.ok.length} pendientes en ${mesNombre(mes)}</div>`;
      } else {
        html += `<div class="rec-summary">Todo al dia en ${mesNombre(mes)}</div>`;
      }

      container.innerHTML = html;
    } catch (e) {
      // Silencioso si el worker no responde
    }
  }

  function mesNombre(m) {
    return ['','Ene','Feb','Mar','Abr','May','Jun','Jul','Ago','Sep','Oct','Nov','Dic'][m] || '';
  }
